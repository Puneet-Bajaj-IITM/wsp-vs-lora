"""Job 1: base model on 2048 random train examples, capture answer-position residuals."""
import json
from pathlib import Path

import numpy as np
import torch

from wsp.harness import Lab

WORK = Path(__file__).resolve().parent
SEED = 20261120
N_PASSES = 4
PASS_SIZE = 512
CAPTURE = (4, 8, 12, 14, 16, 18, 20, 22, 23, 24, 25, 26, -1)
LETTER_STRINGS = ("A", "B", "C", "D", "E", " A", " B", " C", " D", " E")

lab = Lab(edit=None)
cfg = lab.model.config
print("config", cfg.num_hidden_layers, cfg.hidden_size, cfg.num_attention_heads, cfg.num_key_value_heads,
      cfg.intermediate_size, "tied", lab.model.lm_head.weight is lab.model.model.embed_tokens.weight)
train = lab.examples("train")
perm = np.random.default_rng(SEED).permutation(len(train))
idx = perm[: N_PASSES * PASS_SIZE]
logps, hidden = [], {layer: [] for layer in CAPTURE}
for p in range(N_PASSES):
    chunk = [int(i) for i in idx[p * PASS_SIZE:(p + 1) * PASS_SIZE]]
    r = lab.forward("train", chunk, note=f"base train capture {p}", capture_layers=CAPTURE)
    logps.append(r["logp"])
    for layer in CAPTURE:
        hidden[layer].append(r["hidden"][layer].astype(np.float16))
    print(p, json.dumps(r["summary"]), r["passes_left"], flush=True)

W = lab.model.lm_head.weight
letter_ids = [lab.tok.encode(s, add_special_tokens=False) for s in LETTER_STRINGS]
print("letter ids", letter_ids)
flat_ids = [i[0] for i in letter_ids]
out = {f"h{layer}": np.concatenate(v) for layer, v in hidden.items()}
out["idx"] = idx
out["logp"] = np.concatenate(logps)
out["W_letters"] = W[flat_ids].float().cpu().numpy()
out["norm_w"] = lab.model.model.norm.weight.float().cpu().numpy()
out["embed_row_norm_mean"] = np.array([W.float().norm(dim=1).mean().item()])
np.savez(WORK / "job1_base.npz", **out)

ex = train[int(idx[0])]
from wsp.scoring import render_prompts
ids = lab.tok(render_prompts(lab.tok, [ex])[0])["input_ids"]
toks = lab.tok.convert_ids_to_tokens(ids)
print("n tokens", len(toks))
print("HEAD", list(zip(ids[:30], toks[:30])))
print("TAIL", list(zip(ids[-90:], toks[-90:])))
for name, prm in lab.model.model.layers[27].named_parameters():
    print(name, tuple(prm.shape))
